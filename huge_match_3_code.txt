"use client";
import { useState, useRef } from "react";
import Image from "next/image";

export default function Home() {
  const [isRecording, setIsRecording] = useState(false);
  const [imagePreview, setImagePreview] = useState(null);

  const handleImageChange = (e) => {
    const file = e.target.files[0];
    if (file) {
      // Create a local URL for the image to preview it
      const imageUrl = URL.createObjectURL(file);
      setImagePreview(imageUrl);
      console.log("Image selected:", file.name);
    }
  };

  const handleMicClick = () => {
    if (!isRecording) {
      console.log("Started listening...");
      setIsRecording(true);
    } else {
      console.log("Stopped listening.");
      setIsRecording(false);
    }
  };

  return (
    <main className="flex flex-col min-h-screen bg-background font-sans">
      
      {/* Header */}
      <header className="bg-gradient-to-r from-agri-green via-emerald-800 to-agri-green text-white p-5 shadow-lg flex justify-between items-center rounded-b-3xl relative overflow-hidden">
        <div className="pointer-events-none absolute top-0 right-0 w-32 h-32 bg-agri-accent rounded-full mix-blend-screen filter blur-3xl opacity-30 animate-pulse"></div>
        <div className="flex items-center gap-3 relative z-10 pointer-events-none">
          <div className="w-12 h-12 bg-white/10 backdrop-blur-md border border-white/20 rounded-full flex items-center justify-center font-bold text-2xl shadow-[0_0_15px_rgba(82,183,136,0.5)]">
            🌱
          </div>
          <h1 className="text-3xl font-extrabold tracking-wide bg-clip-text text-transparent bg-gradient-to-r from-white to-agri-light">Kisan-Dost AI</h1>
        </div>
      </header>

      {/* Main Content Area */}
      <section className="flex-1 flex flex-col items-center justify-center p-6 gap-8">
        
        {/* Instruction Text */}
        <div className="text-center pointer-events-none">
          <h2 className="text-2xl md:text-3xl font-bold text-agri-green mb-2">
            {isRecording ? "Sun raha hoon..." : "Apna Masla Batayen"}
          </h2>
          <p className="text-gray-600 text-lg">
            {isRecording ? "Bolna khatam karne ke liye dobara dabayen" : "Neechay button daba kar apni zaban mein baat karein."}
          </p>
        </div>

        {/* Action Buttons */}
        <div className="flex flex-col gap-6 w-full max-w-sm relative z-20">
          
          {/* AI Glowing Background for Mic Button */}
          <div className={`pointer-events-none absolute top-0 left-0 w-full h-32 rounded-full mix-blend-multiply filter blur-2xl opacity-30 animate-pulse ${isRecording ? 'bg-red-500 scale-110' : 'bg-agri-accent'}`}></div>
          
          {/* Big Mic Button with AI Vibe */}
          <button 
            type="button"
            onClick={handleMicClick}
            className={`cursor-pointer relative z-30 flex flex-col items-center justify-center gap-3 rounded-3xl p-8 shadow-[0_10px_30px_rgba(27,67,50,0.4)] transition-all duration-300 transform active:scale-90 group
              ${isRecording 
                ? 'bg-gradient-to-b from-red-500 to-red-800 border-red-400 scale-105' 
                : 'bg-gradient-to-b from-agri-green to-emerald-900 hover:from-agri-green hover:to-agri-green border border-emerald-700/50 hover:scale-105'
              }`}
          >
            <div className="relative pointer-events-none">
              <div className="pointer-events-none absolute inset-0 bg-white opacity-0 group-hover:opacity-20 rounded-full blur-md transition-opacity duration-300"></div>
              {isRecording ? (
                 <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" strokeWidth={2} stroke="currentColor" className="w-16 h-16 text-white relative z-10 animate-bounce">
                   <path strokeLinecap="round" strokeLinejoin="round" d="M21 12a9 9 0 11-18 0 9 9 0 0118 0z" />
                   <path strokeLinecap="round" strokeLinejoin="round" d="M9 9.563C9 9.252 9.252 9 9.563 9h4.874c.311 0 .563.252.563.563v4.874c0 .311-.252.563-.563.563H9.564A.562.562 0 019 14.437V9.564z" />
                 </svg>
              ) : (
                <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" strokeWidth={1.5} stroke="currentColor" className="w-16 h-16 text-white relative z-10">
                  <path strokeLinecap="round" strokeLinejoin="round" d="M12 18.75a6 6 0 006-6v-1.5m-6 7.5a6 6 0 01-6-6v-1.5m6 7.5v3.75m-3.75 0h7.5M12 15.75a3 3 0 01-3-3V4.5a3 3 0 116 0v8.25a3 3 0 01-3 3z" />
                </svg>
              )}
            </div>
            <span className="text-2xl font-bold tracking-wide text-white pointer-events-none">
              {isRecording ? "Stop Recording" : "Bol Kar Batayen"}
            </span>
          </button>

          {/* Native HTML Label wrapper for Camera Button to ensure it works on all mobile browsers */}
          <label className="cursor-pointer flex items-center justify-center gap-3 bg-white border-2 border-agri-green text-agri-green hover:bg-agri-light rounded-2xl p-4 shadow-sm transition-transform transform hover:scale-105 active:scale-90 relative z-30">
            <input 
              type="file" 
              accept="image/*" 
              capture="environment" 
              onChange={handleImageChange} 
              className="hidden" 
            />
            <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" strokeWidth={1.5} stroke="currentColor" className="w-8 h-8 pointer-events-none">
              <path strokeLinecap="round" strokeLinejoin="round" d="M6.827 6.175A2.31 2.31 0 015.186 7.23c-.38.054-.757.112-1.134.175C2.999 7.58 2.25 8.507 2.25 9.574V18a2.25 2.25 0 002.25 2.25h15A2.25 2.25 0 0021.75 18V9.574c0-1.067-.75-1.994-1.802-2.169a47.865 47.865 0 00-1.134-.175 2.31 2.31 0 01-1.64-1.055l-.822-1.316a2.192 2.192 0 00-1.736-1.039 48.774 48.774 0 00-5.232 0 2.192 2.192 0 00-1.736 1.039l-.821 1.316z" />
              <path strokeLinecap="round" strokeLinejoin="round" d="M16.5 12.75a4.5 4.5 0 11-9 0 4.5 4.5 0 019 0zM18.75 10.5h.008v.008h-.008V10.5z" />
            </svg>
            <span className="text-lg font-bold pointer-events-none">Tasweer Khinchein</span>
          </label>
        </div>
        
        {/* Image Preview Area */}
        {imagePreview && (
          <div className="mt-4 flex flex-col items-center relative z-20">
            <p className="text-sm font-bold text-agri-green mb-2">Aap ki tasweer:</p>
            <div className="relative w-48 h-48 rounded-2xl overflow-hidden border-4 border-agri-accent shadow-md">
               <img src={imagePreview} alt="Crop Preview" className="object-cover w-full h-full" />
            </div>
            <button 
              onClick={() => setImagePreview(null)}
              className="mt-2 text-red-500 text-sm font-bold cursor-pointer"
            >
              Tasweer Hataen (Cancel)
            </button>
          </div>
        )}

      </section>

      {/* Language Selector Footer */}
      <footer className="bg-white p-4 shadow-inner mt-auto relative z-30">
        <p className="text-center text-sm text-gray-500 mb-2">Zaban Muntakhib Karein (Select Language)</p>
        <div className="flex justify-center gap-4">
          <button type="button" className="cursor-pointer px-4 py-2 bg-agri-green text-white rounded-full font-bold shadow hover:bg-agri-green/90 transition">🇵🇰 Urdu</button>
          <button type="button" className="cursor-pointer px-4 py-2 bg-gray-100 text-gray-700 rounded-full font-bold shadow hover:bg-gray-200 transition border border-gray-200">🌾 Punjabi</button>
          <button type="button" className="cursor-pointer px-4 py-2 bg-gray-100 text-gray-700 rounded-full font-bold shadow hover:bg-gray-200 transition border border-gray-200">🐪 Sindhi</button>
        </div>
      </footer>
    </main>
  );
}
